#! /bin/sh
# Script showing how to use md1D using the most classical command line arguments

# Path to the program (using the environment variable $HOME, defined by the system)
dir=$HOME"/Programmes/md1D"

# Most md1D options can be set using command-line arguments
# If a parameter is indicated both in the param_file and by a command-line 
# argument, the command-line argument is prioritary.
# For more information on the parameters see "param_01_basic.txt" or the param-file
# in this directory

# Calling md1D with 3 command-line arguments
$dir/md1D -L 1000.0 -N 10 -param param_02_command-line.txt

# Calling md1D with several command-line arguments
$dir/md1D -L 3000.0 -h 10.0 -coef_h 2.0 -N 15 -NS 10 \
          -lambda 632.8 -angle_i 50.0 -pola TE \
	  -delta_h 0.5 -param param_02_command-line.txt \
          -fichier_profil cos_256.dat -nom_profil results_toto


# The values of the command-line arguments can be set directly or by using a
# variable, which is generaly more interesting because it can be calculated
# or modified by the script

# Setting parameters values 
L=1000
h=1.0
coef_h=2.0
N=15
NS=10
lambda=632.8
angle_i=50.0
pola="TE"
delta_h=0.5
pr_file="cos_256.dat"
pr_name="results_toto"
param_file="param_02_command-line.txt"

# Calling md1D with command-line options
$dir/md1D -L $L -h $h -coef_h $coef_h -N $N -NS $NS \
          -lambda $lambda -angle_i $angle_i -pola $pola \
	  -delta_h $delta_h -param $param_file \
          -fichier_profil $pr_file -nom_profil $pr_name

# For more detailed informations on command-line arguments take a look at md1D_in_out.c
